cmake_minimum_required(VERSION 3.20)
project(OpusRuntime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
set(CT2_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../Dependencies/vendor/CTranslate2" CACHE PATH "")
set(SPM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../Dependencies/vendor/sentencepiece" CACHE PATH "")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_ACCELERATE ON CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_MOCK_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
# SentencePiece's iOS CMake expects this helper from ios-cmake even when using
# CMake's built-in Apple cross compilation. Its tools are not built by this target.
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
 set(IOS ON)
 set(IOS_ARCH arm64)
 function(set_xcode_property target property value variant)
  set_target_properties(${target} PROPERTIES XCODE_ATTRIBUTE_${property} "${value}")
 endfunction()
endif()
add_subdirectory("${CT2_SOURCE}" ct2 EXCLUDE_FROM_ALL)
add_subdirectory("${SPM_SOURCE}" sentencepiece EXCLUDE_FROM_ALL)
add_library(OpusRuntime SHARED OpusRuntime.cpp)
target_include_directories(OpusRuntime PRIVATE "${CT2_SOURCE}/include" "${CT2_SOURCE}/third_party" "${SPM_SOURCE}/src")
target_link_libraries(OpusRuntime PRIVATE ctranslate2 sentencepiece-static "-framework Accelerate")
target_link_options(OpusRuntime PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt")
set_target_properties(OpusRuntime PROPERTIES
 FRAMEWORK TRUE FRAMEWORK_VERSION A
 MACOSX_FRAMEWORK_BUNDLE_VERSION 1
 MACOSX_FRAMEWORK_SHORT_VERSION_STRING 1.0
 MACOSX_FRAMEWORK_IDENTIFIER org.openoffline.OpusRuntime
 PUBLIC_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/OpusRuntime.h")
