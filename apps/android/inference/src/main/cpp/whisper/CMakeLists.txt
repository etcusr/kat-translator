cmake_minimum_required(VERSION 3.22.1)
project(kat_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_COREML GGML_METAL GGML_BLAS GGML_ACCELERATE GGML_OPENMP GGML_NATIVE GGML_VULKAN GGML_BACKEND_DL GGML_CPU_KLEIDIAI)
 set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
get_filename_component(KAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../.." ABSOLUTE)
option(KAT_WHISPER_ARM82 "Build the optional FP16/dotprod Whisper library" OFF)
option(KAT_WHISPER_I8MM "Build the optional Q8 i8mm Whisper library" OFF)
if(KAT_WHISPER_I8MM)
 set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16+i8mm" CACHE STRING "" FORCE)
 set(KAT_WHISPER_TARGET kat_whisper_i8mm)
elseif(KAT_WHISPER_ARM82)
 set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
 set(KAT_WHISPER_TARGET kat_whisper_arm82)
else()
 # Always keep the baseline usable on older arm64 phones, including reused build trees.
 set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
 set(KAT_WHISPER_TARGET kat_whisper)
endif()
add_subdirectory("${KAT_ROOT}/Dependencies/vendor/whisper.cpp" whisper)
# Forward cancellation into GGML's scheduler on Android only. The shared pinned
# vendor checkout is also built by Apple and must remain untouched.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(KAT_WHISPER_SOURCE "${KAT_ROOT}/Dependencies/vendor/whisper.cpp/src/whisper.cpp")
set(KAT_WHISPER_PATCH "${KAT_ROOT}/scripts/android/patch-whisper-cancellation.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KAT_WHISPER_SOURCE}" "${KAT_WHISPER_PATCH}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${KAT_WHISPER_PATCH}" "${KAT_WHISPER_SOURCE}"
 "${CMAKE_CURRENT_BINARY_DIR}/whisper-cancellable.cpp" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(KAT_WHISPER_SOURCES whisper SOURCES)
list(REMOVE_ITEM KAT_WHISPER_SOURCES whisper.cpp)
set_property(TARGET whisper PROPERTY SOURCES ${KAT_WHISPER_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/whisper-cancellable.cpp")
target_include_directories(whisper PRIVATE "${KAT_ROOT}/Dependencies/vendor/whisper.cpp/src")
add_library(${KAT_WHISPER_TARGET} SHARED ../whisper_jni.cpp "${KAT_ROOT}/shared/native/app/ASR/WhisperCancellation.c")
if(KAT_WHISPER_I8MM)
 target_compile_definitions(${KAT_WHISPER_TARGET} PRIVATE KAT_WHISPER_I8MM)
elseif(KAT_WHISPER_ARM82)
 target_compile_definitions(${KAT_WHISPER_TARGET} PRIVATE KAT_WHISPER_ARM82)
endif()
target_include_directories(${KAT_WHISPER_TARGET} PRIVATE "${KAT_ROOT}/shared/native/app")
target_link_libraries(${KAT_WHISPER_TARGET} PRIVATE whisper android log)
target_link_options(${KAT_WHISPER_TARGET} PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/../media-exports.map" "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
