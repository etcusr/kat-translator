cmake_minimum_required(VERSION 3.22.1)
project(kat_omnivoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt GGML_BUILD_TESTS GGML_BUILD_EXAMPLES GGML_METAL GGML_BLAS GGML_ACCELERATE GGML_OPENMP GGML_NATIVE GGML_VULKAN GGML_BACKEND_DL)
 set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
get_filename_component(KAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../.." ABSOLUTE)
set(OV_SOURCE "${KAT_ROOT}/Dependencies/vendor/omnivoice-mobile")
add_compile_definitions(GGML_MAX_NAME=128)
add_subdirectory("${OV_SOURCE}/ggml" ggml)
# Preserve native SDOT lane semantics on baseline ARM64 without requiring the
# optional DOTPROD extension. Applies only inside the isolated OmniVoice DSO.
target_compile_options(ggml-cpu PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/../omnivoice_cpu_compat.h")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h" "#pragma once\n#define OMNIVOICE_VERSION \"53e6c206-android\"\n")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OV_ANDROID_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/android-diagnostics")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KAT_ROOT}/scripts/android/prepare-omnivoice-diagnostics.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${KAT_ROOT}/scripts/android/prepare-omnivoice-diagnostics.py"
 "${OV_SOURCE}/src" "${OV_ANDROID_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
add_library(kat_omnivoice SHARED ../omnivoice_jni.cpp "${OV_SOURCE}/src/omnivoice.cpp" "${OV_ANDROID_SOURCE}/pipeline-codec.cpp" "${OV_ANDROID_SOURCE}/pipeline-tts.cpp" "${KAT_ROOT}/shared/native/app/ASR/WhisperCancellation.c")
target_compile_definitions(kat_omnivoice PRIVATE OMNIVOICE_BUILD)
target_include_directories(kat_omnivoice PRIVATE "${OV_SOURCE}/src" "${OV_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}" "${KAT_ROOT}/shared/native/app")
target_link_libraries(kat_omnivoice PRIVATE ggml ggml-base ggml-cpu android log)
target_link_options(kat_omnivoice PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/../media-exports.map" "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
