cmake_minimum_required(VERSION 3.22.1)
project(kat_index LANGUAGES C CXX)
get_filename_component(KAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_APP LLAMA_BUILD_COMMON LLAMA_BUILD_SERVER LLAMA_BUILD_MTMD LLAMA_CURL GGML_METAL GGML_BLAS GGML_ACCELERATE GGML_OPENMP GGML_NATIVE GGML_BACKEND_DL)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
option(KAT_GPU_DIAGNOSTICS "Build standalone on-device GPU probes (not packaged in APK)" OFF)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
if(KAT_GPU_DIAGNOSTICS)
    set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
    set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
    set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
endif()
# Without an explicit arch the NDK targets plain armv8-a: no dotprod/fp16, so the
# CPU backend ran ~2.4x slower on Pixel 9 Pro (2026-10-12 audit). i8mm is left
# out: Tensor G1/G2 and Snapdragon 865/888 lack it and the measured gain was
# within noise. index_jni.cpp checks HWCAP before any ggml code runs.
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(KAT_VK_TOOLS "${KAT_ROOT}/.build/android-tools/vulkan")
set(Vulkan_GLSLC_EXECUTABLE "${KAT_VK_TOOLS}/shaderc/2026.4/bin/glslc" CACHE FILEPATH "")
set(Vulkan_INCLUDE_DIR "${KAT_VK_TOOLS}/vulkan-headers/1.4.363.0/include" CACHE PATH "")
find_library(KAT_VULKAN_LIBRARY vulkan REQUIRED)
set(Vulkan_LIBRARY "${KAT_VULKAN_LIBRARY}" CACHE FILEPATH "")
set(SPIRV-Headers_DIR "${KAT_VK_TOOLS}/spirv-headers/1.4.363.0/share/cmake/SPIRV-Headers" CACHE PATH "")
if(NOT EXISTS "${Vulkan_GLSLC_EXECUTABLE}")
    message(FATAL_ERROR "Run scripts/android/setup-vulkan-tools.py before building")
endif()
add_subdirectory("${KAT_ROOT}/Dependencies/vendor/llama.cpp" llama)
include("${CMAKE_CURRENT_SOURCE_DIR}/mali-narrow-tiles.cmake")
target_include_directories(ggml-vulkan PRIVATE "${KAT_VK_TOOLS}/spirv-headers/1.4.363.0/include")
add_library(kat_index SHARED index_jni.cpp index_mtp.cpp memory_probe.cpp gpu_probe_matvec.cpp "${KAT_ROOT}/shared/native/TranslationRuntime/TranslationRuntime.cpp")
target_include_directories(kat_index PRIVATE "${KAT_ROOT}/shared/native/TranslationRuntime")
target_link_libraries(kat_index PRIVATE llama llama-common android log)
target_link_options(kat_index PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
if(KAT_GPU_DIAGNOSTICS)
    include("${CMAKE_CURRENT_SOURCE_DIR}/gpu-diagnostics.cmake")
    add_executable(kat_gpu_probe "${KAT_ROOT}/scripts/android/native/gpu_probe.cpp")
    target_link_libraries(kat_gpu_probe PRIVATE ggml android log)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/media.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/omnivoice.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/wayu.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/opus.cmake")
