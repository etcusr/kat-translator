cmake_minimum_required(VERSION 3.22.1)
project(kat_opus_gpu_probe LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/Dependencies/vendor/sentencepiece" spm EXCLUDE_FROM_ALL)
set(SRC "${ROOT}/.build/opus-vulkan-lab/src")
add_executable(opus-gpu-probe main.cpp "${SRC}/model.cpp" "${SRC}/tokenizer.cpp" "${SRC}/encoder.cpp" "${SRC}/decoder.cpp")
target_include_directories(opus-gpu-probe PRIVATE "${SRC}" "${ROOT}/Dependencies/vendor/llama.cpp/ggml/include" "${ROOT}/Dependencies/vendor/sentencepiece/src")
set(LIB "${ROOT}/.build/mali-kernels/native/llama/ggml/src")
target_link_libraries(opus-gpu-probe PRIVATE "${LIB}/libggml.a" "${LIB}/ggml-vulkan/libggml-vulkan.a" "${LIB}/libggml-cpu.a" "${LIB}/libggml-base.a" sentencepiece-static vulkan log android)
target_link_options(opus-gpu-probe PRIVATE "-Wl,-z,max-page-size=16384")
set(CT2 "${ROOT}/Dependencies/vendor/CTranslate2")
file(GLOB_RECURSE CT2_LIBS "${ROOT}/apps/android/inference/.cxx/RelWithDebInfo/39l21456/arm64-v8a/opus-ct2/*.a")
add_executable(opus-cpu-control cpu-main.cpp "${ROOT}/shared/native/OpusRuntime/OpusRuntime.cpp")
target_include_directories(opus-cpu-control PRIVATE "${ROOT}/shared/native/OpusRuntime" "${CT2}/include" "${CT2}/third_party" "${ROOT}/Dependencies/vendor/sentencepiece/src")
target_link_libraries(opus-cpu-control PRIVATE -Wl,--start-group ${CT2_LIBS} sentencepiece-static -Wl,--end-group log android)
