cmake_minimum_required(VERSION 3.20)
project(WhisperAcceleration LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "Pass WHISPER_SOURCE pointing to the pinned vendor checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML ON CACHE BOOL "" FORCE)
set(WHISPER_COREML_ALLOW_FALLBACK ON CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_BLAS ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(whisper-bench Bench.mm CoreMLBridge.mm)
  target_include_directories(whisper-bench PRIVATE "${WHISPER_SOURCE}/src/coreml")
  target_compile_options(whisper-bench PRIVATE -fobjc-arc)
  target_link_libraries(whisper-bench PRIVATE whisper "-framework Foundation" "-framework Metal" "-framework CoreML")
endif()
